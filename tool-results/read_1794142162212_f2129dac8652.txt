   194→/**
   195→ * Local board selection for learn-mode boards (tap-to-move + drag reveal).
   196→ * Mirrors the play-room selection semantics against a read-only FEN.
   197→ */
   198→function useBoardSelection(
   199→  fen: string,
   200→  onMove: (from: Square, to: Square) => boolean
   201→) {
   202→  const [selected, setSelected] = useState<Square | null>(null);
   203→  const [legal, setLegal] = useState<Move[]>([]);
   204→
   205→  const select = useMemo(
   206→    () =>
   207→      (sq: Square | null) => {
   208→        if (!sq || selected === sq) {
   209→          setSelected(null);
   210→          setLegal([]);
   211→          return;
   212→        }
   213→        const g = new Chess(fen);
   214→        const piece = g.get(sq);
   215→        if (piece && piece.color === g.turn()) {
   216→          setSelected(sq);
   217→          setLegal(g.moves({ square: sq, verbose: true }));
   218→          return;
   219→        }
   220→        if (selected && legal.some((m) => m.to === sq)) {
   221→          if (onMove(selected, sq)) {
   222→            setSelected(null);
   223→            setLegal([]);
   224→          }
   225→          return;
   226→        }
   227→        setSelected(null);
   228→        setLegal([]);
   229→      },
   230→    [selected, legal, fen, onMove]
   231→  );
   232→
   233→  return { selected, legal, select };
   234→}

   443→function puzzleStepText(pz: PzHook): string {
   444→  const line = pz.puzzle.lines[0];
   445→  const total = line.userMoves.length;
   446→  if (total === 1) return "One move wins. Find it and play it on the board.";
   447→  return `Find the winning sequence — move ${Math.min(pz.step + 1, total)} of ${total}. Take your time.`;
   448→}

   452→function DailyPanel() {
   453→  const pz = usePuzzles({ daily: true });
   454→  const career = useCareer();
   455→  const conqueredToday = career.dailyDoneDate === dateKey();
   456→
   457→  // countdown to midnight, refreshed every 30s
   458→  const [now, setNow] = useState(() => Date.now());
   459→  useEffect(() => {
   460→    const iv = window.setInterval(() => setNow(Date.now()), 30_000);
   461→    return () => window.clearInterval(iv);
   462→  }, []);
   463→
   464→  const countdown = useMemo(() => {
   465→    const end = new Date();
   466→    end.setHours(24, 0, 0, 0);
   467→    const ms = Math.max(0, end.getTime() - now);
   468→    const h = Math.floor(ms / 3_600_000);
   469→    const m = Math.floor((ms % 3_600_000) / 60_000);
   470→    return `${h}h ${m}m`;
   471→  }, [now]);
   472→
   473→  const done = pz.status === "solved" || conqueredToday;
   474→
   475→  return (
   476→    <div className="space-y-4">
   477→      <div
   478→        className={cn(
   479→          "glass-panel float-in relative overflow-hidden rounded-2xl p-4 sm:p-5",
