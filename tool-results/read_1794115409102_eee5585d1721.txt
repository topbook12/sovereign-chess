    85→function loadJSON<T>(key: string, fallback: T): T {
    86→  if (typeof window === "undefined") return fallback;
    87→  try {
    88→    const raw = window.localStorage.getItem(key);
    89→    return raw ? { ...fallback, ...(JSON.parse(raw) as object) } as T : fallback;
    90→  } catch {
    91→    return fallback;
    92→  }
    93→}

    95→function sideOf(c: Color): Side {
    96→  return c === "w" ? "white" : "black";
    97→}
    98→function charOf(s: Side): Color {
    99→  return s === "white" ? "w" : "b";

   102→function fenToPieces(fen: string, opts?: { spawn?: boolean }): PieceView[] {
   103→  const placement = fen.split(" ")[0];
   104→  const rows = placement.split("/");
   105→  const out: PieceView[] = [];
   106→  let n = 0;
   107→  for (let r = 0; r < 8; r++) {
   108→    let file = 0;
   109→    for (const ch of rows[r]) {
   110→      if (/\d/.test(ch)) {
   111→        file += Number(ch);
   112→        continue;
   113→      }
   114→      const color: Color = ch === ch.toUpperCase() ? "w" : "b";
   115→      const type = ch.toLowerCase() as PieceSymbol;
   116→      const rank = 8 - r;
   117→      const square = `${String.fromCharCode(97 + file)}${rank}` as Square;
   118→      out.push({
   119→        id: `p${n}-${color}${type}-${square}`,
   120→        square,
   121→        type,
   122→        color,
   123→        spawn: opts?.spawn,
   124→        spawnDelay: opts?.spawn ? n * 14 : 0,
   125→      });
   126→      n++;
   127→      file++;
   128→    }
   129→  }
   130→  return out;
   131→}
   132→
   133→function classifyDraw(game: Chess): GameResult["reason"] {
   134→  if (game.isStalemate()) return "stalemate";
   135→  if (game.isThreefoldRepetition()) return "threefold";
   136→  if (game.isInsufficientMaterial()) return "insufficient";
   137→  return "fifty";
   138→}

   232→    const desc = (a: PieceSymbol, b: PieceSymbol) => val(b) - val(a);

   234→    byBlack.sort(desc);

   236→    const materialBlack = byBlack.reduce((s, p) => s + val(p), 0);

   274→  }, []);
   275→
   276→  useEffect(() => {
   277→    try {
   278→      window.localStorage.setItem(
   279→        SETTINGS_KEY,
   280→        JSON.stringify({
   281→          mode,
   282→          difficulty,
   283→          timeControlId: activeTimeControl.id,
   284→          side: preferredSide,
   285→          sound: soundOn,

   287→          showCoords,

   290→      );
   291→    } catch {

   293→    }
   294→  }, [
   295→    mode,
   296→    difficulty,
   297→    activeTimeControl.id,
   298→    preferredSide,

   303→  ]);
   304→
   305→  useEffect(() => {
   306→    setStats(loadJSON<Stats>(STATS_KEY, { wins: 0, losses: 0, draws: 0 }));

   308→
   309→  const bumpStats = useCallback((r: GameResult) => {
   310→    setStats((prev) => {
   311→      const next = { ...prev };
   312→      if (!r.winner) next.draws++;
   313→      else if (r.winner === humanColor) next.wins++;
   314→      else next.losses++;
   315→      try {
   316→        window.localStorage.setItem(STATS_KEY, JSON.stringify(next));
   317→      } catch {
   318→        /* ignore */
   319→      }
   320→      return next;
   321→    });
   322→  }, [humanColor]);
   323→
   324→  // ---------- helpers ----------
   325→  const refreshBoardState = useCallback((rebuildPieces = true) => {
   326→    const g = gameRef.current;
   327→    setFen(g.fen());
   328→    setTurn(g.turn());
   329→    const tail = g.history({ verbose: true });
   330→    setHistory(tail);

   332→    setLastMove(lm ? { from: lm.from, to: lm.to } : null);
   333→    const check = g.isCheck();
   334→    setInCheck(check);
   335→    if (check) {
   336→      const board = g.board();
   337→      let kingSq: Square | null = null;
   338→      for (const row of board) {
   339→        for (const sq of row) {
   340→          if (sq && sq.type === "k" && sq.color === g.turn()) {
   341→            kingSq = sq.square;
   342→          }
   343→        }
   344→      }
   345→      setCheckedSquare(kingSq);

   350→      setPieces(fenToPieces(g.fen()));

   353→
   354→  const endGame = useCallback(
   355→    (r: GameResult, opts?: { silent?: boolean; countStats?: boolean }) => {
   356→      if (gameRef.current.isGameOver() === false && r.reason === "checkmate") return;

   358→      setThinking(false);
   359→      setStatus("ended");
   360→      setResult(r);
   361→      setSelected(null);
   362→      setLegalForSelected([]);
   363→      setPendingPromotion(null);
   364→      setHint(null);

   366→        if (!r.winner) playSound("draw");
   367→        else if (mode === "human") playSound("win");
   368→        else if (r.winner === humanColor) playSound("win");
   369→        else playSound("lose");
   370→      }
   371→      if (opts?.countStats !== false && mode === "ai") bumpStats(r);
   372→    },
   373→    [mode, humanColor, bumpStats]
   374→  );
   375→
   376→  const applyMoveRef = useRef<(mv: {
   377→    from: Square;
   378→    to: Square;
   379→    promotion?: PromotionPiece;
   380→  }) => boolean>(() => false);
   381→
   382→  // ---------- engine ----------
   383→  const requestEngineMove = useCallback(
   384→    async (reqId: number) => {
   385→      const g = gameRef.current;
   386→      setThinking(true);
   387→      const t0 = Date.now();
   388→      try {
   389→        const res = await fetch("/api/engine", {
   390→          method: "POST",
   391→          headers: { "Content-Type": "application/json" },
   392→          body: JSON.stringify({ fen: g.fen(), level: difficulty }),
   393→        });
   394→        if (!res.ok) throw new Error(`Engine error ${res.status}`);
   395→        const data = (await res.json()) as {
   396→          from: Square;
   397→          to: Square;
   398→          promotion?: string;
   399→          san: string;
   400→          evalCp: number;
   401→          mateIn?: number;
   402→          depth: number;
   403→          nodes: number;
   404→          timeMs: number;
   405→        };
   406→        if (
   407→          reqId !== engineReqRef.current ||
   408→          status !== "active" ||
   409→          gameRef.current.fen() !== g.fen()
   410→        ) {
   411→          return; // stale
   412→        }
   413→        // reveal the move with a brief arrow, then play it
   414→        setHint({ from: data.from, to: data.to });
   415→        const wait = Math.max(180, 600 - (Date.now() - t0));
   416→        await new Promise((r) => setTimeout(r, wait));
   417→        if (reqId !== engineReqRef.current || status !== "active") return;
   418→        setHint(null);
   419→        applyMoveRef.current({
   420→          from: data.from,
   421→          to: data.to,
   422→          promotion: (data.promotion as PromotionPiece) ?? "q",
   423→        });
   424→        setEngineInfo({
   425→          evalCp: data.evalCp,
   426→          mateIn: data.mateIn,
   427→          depth: data.depth,
   428→          nodes: data.nodes,
   429→          timeMs: data.timeMs,
   430→          level: difficulty,
   431→        });
   432→      } catch {
   433→        if (reqId === engineReqRef.current) {
   434→          setThinking(false);
   435→          toast.error("The engine stalled — try again or start a new game.");
   436→        }
   437→      } finally {
   438→        if (reqId === engineReqRef.current) setThinking(false);
   439→      }
   440→    },
   441→    [difficulty, status]
   442→  );
   443→
   444→  // ---------- moving ----------
   445→  const applyMove = useCallback(
   446→    (mv: { from: Square; to: Square; promotion?: PromotionPiece }): boolean => {
   447→      const g = gameRef.current;
   448→      let move: Move;
   449→      try {
   450→        move = g.move({ from: mv.from, to: mv.to, promotion: mv.promotion ?? "q" });
   451→      } catch {
   452→        playSound("illegal");
   453→        return false;
   454→      }
   455→
   456→      idRef.current++;
   457→      // reconcile animated pieces
   458→      setPieces((prev) => {
   459→        const next: PieceView[] = [];
   460→        const epSquare = move.flags.includes("e")
   461→          ? (`${move.to[0]}${move.from[1]}` as Square)
   462→          : null;
   463→        for (const p of prev) {
   464→          if (p.captured) continue; // already fading
   465→          if (epSquare && p.square === epSquare && p.color !== move.color) {
   466→            next.push({ ...p, id: `${p.id}-x${idRef.current}`, captured: true });
   467→            continue;
   468→          }
   469→          if (p.square === move.to && p.color !== move.color) {
   470→            next.push({ ...p, id: `${p.id}-x${idRef.current}`, captured: true });
   471→            continue;
   472→          }
   473→          if (p.square === move.from && p.color === move.color && p.type === move.piece) {
   474→            next.push({
   475→              ...p,
   476→              square: move.to,
   477→              type: move.promotion ?? p.type,
   478→            });
   479→            continue;
   480→          }
   481→          // castling rook
   482→          if (
   483→            move.piece === "k" &&
   484→            p.type === "r" &&
   485→            p.color === move.color &&
   486→            (move.flags.includes("k") &&
   487→              p.square === (move.color === "w" ? "h1" : "h8"))
   488→          ) {
   489→            next.push({
   490→              ...p,
   491→              square: (move.color === "w" ? "f1" : "f8") as Square,
   492→            });
   493→            continue;
   494→          }
   495→          if (
   496→            move.piece === "k" &&
   497→            p.type === "r" &&
   498→            p.color === move.color &&
   499→            (move.flags.includes("q") &&
   500→              p.square === (move.color === "w" ? "a1" : "a8"))
   501→          ) {
   502→            next.push({
   503→              ...p,
   504→              square: (move.color === "w" ? "d1" : "d8") as Square,
   505→            });
   506→            continue;
   507→          }
   508→          next.push(p);
   509→        }
   510→        return next;
   511→      });
   512→
   513→      // cleanup captured fade-out pieces
   514→      window.setTimeout(() => {
   515→        setPieces((prev) => prev.filter((p) => !p.captured));
   516→      }, 320);
   517→
   518→      refreshBoardState(false);
   519→
   520→      // sounds
   521→      if (move.flags.includes("k") || move.flags.includes("q")) playSound("castle");
   522→      else if (move.captured) playSound("capture");
   523→      else if (move.promotion) playSound("promote");
   524→      else playSound("move");
   525→
   526→      const gg = gameRef.current;
   527→      if (gg.isCheckmate()) {
   528→        window.setTimeout(
   529→          () =>
   530→            endGame({
   531→              winner: sideOf(gg.turn() === "w" ? "b" : "w"),
   532→              reason: "checkmate",
   533→            }),
   534→          420
   535→        );
   536→      } else if (gg.isDraw()) {
   537→        window.setTimeout(() => endGame({ winner: null, reason: classifyDraw(gg) }), 420);
   538→      } else if (gg.isCheck()) {
   539→        window.setTimeout(() => playSound("check"), 90);
   540→      }
   541→
   542→      setSelected(null);
   543→      setLegalForSelected([]);
   544→      setPendingPromotion(null);
   545→      return true;
   546→    },
   547→    [endGame, refreshBoardState]
   548→  );
   549→
   550→  useEffect(() => {
   551→    applyMoveRef.current = applyMove;
   552→  }, [applyMove]);
   553→
   554→  const attemptMove = useCallback(
   555→    (from: Square, to: Square): boolean => {
   556→      if (!interactive) return false;
   557→      const g = gameRef.current;
   558→      const candidates = g
   559→        .moves({ square: from, verbose: true })
   560→        .filter((m) => m.to === to);
   561→      if (candidates.length === 0) {
   562→        playSound("illegal");
   563→        return false;
   564→      }
   565→      if (candidates[0].promotion) {
   566→        if (autoQueen) {
   567→          return applyMove({ from, to, promotion: "q" });
   568→        }
   569→        setPendingPromotion({ from, to });
   570→        return true;
   571→      }
   572→      return applyMove({ from, to });
   573→    },
   574→    [interactive, autoQueen, applyMove]
   575→  );
   576→
   577→  const selectSquare = useCallback(
   578→    (square: Square | null) => {
   579→      if (!interactive) {
   580→        setSelected(null);
   581→        setLegalForSelected([]);
   582→        return;
   583→      }
   584→      if (square === null) {
   585→        setSelected(null);
   586→        setLegalForSelected([]);
   587→        return;
   588→      }
   589→      if (selected === square) {
   590→        setSelected(null);
   591→        setLegalForSelected([]);
   592→        return;
   593→      }
   594→      const g = gameRef.current;
   595→      const piece = g.get(square);
   596→      if (piece && piece.color === g.turn()) {
   597→        setSelected(square);
   598→        setLegalForSelected(g.moves({ square, verbose: true }));
   599→        playSound("select");
   600→        return;
   601→      }
   602→      // clicking a target square
   603→      if (selected && legalForSelected.some((m) => m.to === square)) {
   604→        attemptMove(selected, square);
   605→        return;
   606→      }
   607→      setSelected(null);
   608→      setLegalForSelected([]);
   609→    },
   610→    [interactive, selected, legalForSelected, attemptMove]
   611→  );
   612→
   613→  const promote = useCallback(
   614→    (piece: PromotionPiece) => {
   615→      if (!pendingPromotion) return;
   616→      applyMove({ from: pendingPromotion.from, to: pendingPromotion.to, promotion: piece });
   617→    },
   618→    [pendingPromotion, applyMove]
   619→  );
   620→
   621→  const cancelPromotion = useCallback(() => {
   622→    setPendingPromotion(null);
   623→    setSelected(null);
   624→    setLegalForSelected([]);
   625→  }, []);
   626→
   627→  // ---------- engine auto-play effect ----------
   628→  const engineTurnKey = `${fen}|${mode}|${status}|${viewIndex}|${difficulty}|${pendingPromotion ? 1 : 0}`;
   629→  const lastEngineTurnKey = useRef("");
   630→  useEffect(() => {
   631→    if (mode !== "ai" || status !== "active" || viewing || pendingPromotion) {
   632→      return;
   633→    }
   634→    if (gameRef.current.turn() !== engineColor) return;
   635→    if (lastEngineTurnKey.current === engineTurnKey) return;
   636→    lastEngineTurnKey.current = engineTurnKey;
   637→    const reqId = ++engineReqRef.current;
   638→    const t = window.setTimeout(() => void requestEngineMove(reqId), 240);
   639→    return () => window.clearTimeout(t);
   640→  }, [engineTurnKey]);
   641→
   642→  // ---------- clock ----------
   643→  useEffect(() => {
   644→    clockSideRef.current = turn;
   645→  }, [turn]);
   646→
   647→  useEffect(() => {
   648→    if (status !== "active" || activeTimeControl.initialMs === 0) return;
   649→    if (history.length === 0) return; // clock starts after the first move
   650→    let last = Date.now();
   651→    const iv = window.setInterval(() => {
   652→      const now = Date.now();
   653→      const dt = now - last;
   654→      last = now;
   655→      setClocks((prev) => {
   656→        const side = clockSideRef.current;
   657→        return { ...prev, [side]: Math.max(0, prev[side] - dt) };
   658→      });
   659→    }, 100);
   660→    return () => window.clearInterval(iv);
   661→  }, [status, activeTimeControl.initialMs, turn, fen]);
   662→
   663→  useEffect(() => {
   664→    if (status !== "active" || activeTimeControl.initialMs === 0) return;
   665→    if (clocks.w <= 0) endGame({ winner: "black", reason: "timeout" });
   666→    else if (clocks.b <= 0) endGame({ winner: "white", reason: "timeout" });
   667→  }, [clocks, status, activeTimeControl.initialMs, endGame]);
   668→
   669→  // ---------- actions ----------
   670→  const newGame = useCallback(
   671→    (opts?: {
   672→      mode?: GameMode;
   673→      difficulty?: Difficulty;
   674→      side?: Side | "random";
   675→      timeControlId?: string;
   676→    }) => {
   677→      engineReqRef.current++;
   678→      const nextMode = opts?.mode ?? mode;
   679→      const nextSide: Side =
   680→        opts?.side === "random"
   681→          ? Math.random() < 0.5
   682→            ? "white"
   683→            : "black"
   684→          : (opts?.side ?? preferredSide);
   685→      const tc =
   686→        TIME_CONTROLS.find((t) => t.id === opts?.timeControlId) ??
   687→        activeTimeControl;
   688→
   689→      gameRef.current = new Chess();
   690→      idRef.current++;
   691→      lastEngineTurnKey.current = "";
   692→      setMode(nextMode);
   693→      if (opts?.difficulty) setDifficulty(opts.difficulty);
   694→      if (opts?.side) setPreferredSide(nextSide === "white" ? "white" : "black");
   695→      setActiveTimeControl(tc);
   696→      setHumanColor(nextSide);
   697→      setOrientation(nextSide);
   698→      setFen(gameRef.current.fen());
   699→      setPieces(fenToPieces(gameRef.current.fen(), { spawn: true }));
   700→      setHistory([]);
   701→      setTurn("w");
   702→      setLastMove(null);
   703→      setInCheck(false);
   704→      setCheckedSquare(null);
   705→      setSelected(null);
   706→      setLegalForSelected([]);
   707→      setPendingPromotion(null);
   708→      setStatus("active");
   709→      setResult(null);
   710→      setDrawOfferPending(null);
   711→      setClocks({ w: tc.initialMs, b: tc.initialMs });
   712→      setThinking(false);
   713→      setEngineInfo(null);
   714→      setHint(null);
   715→      setHintsUsed(0);
   716→      setViewIndex(null);
   717→    },
   718→    [mode, preferredSide, activeTimeControl]
   719→  );
   720→
   721→  const undo = useCallback(() => {
   722→    engineReqRef.current++;
   723→    setThinking(false);
   724→    const g = gameRef.current;
   725→    if (g.history().length === 0) return;
   726→    if (mode === "ai" && g.turn() === charOf(humanColor) && g.history().length >= 2) {
   727→      g.undo(); // engine reply
   728→      g.undo(); // human move
   729→    } else {
   730→      g.undo();
   731→    }
   732→    setSelected(null);
   733→    setLegalForSelected([]);
   734→    setPendingPromotion(null);
   735→    setHint(null);
   736→    setViewIndex(null);
   737→    if (status === "ended") {
   738→      setStatus("active");
   739→      setResult(null);
   740→    }
   741→    lastEngineTurnKey.current = "";
   742→    refreshBoardState();
   743→    playSound("move");
   744→  }, [mode, humanColor, status, refreshBoardState]);
   745→
   746→  const resign = useCallback(() => {
   747→    if (status !== "active") return;
   748→    const loser: Side = mode === "ai" ? humanColor : sideOf(turn);
   749→    endGame({ winner: loser === "white" ? "black" : "white", reason: "resignation" });
   750→    toast(`${loser === "white" ? "White" : "Black"} resigned.`, {
   751→      description: "The crown changes hands.",
   752→    });
   753→  }, [status, mode, humanColor, turn, endGame]);
   754→
   755→  const offerDraw = useCallback(() => {
   756→    if (status !== "active") return;
   757→    if (mode === "human") {
   758→      setDrawOfferPending(turn);
   759→      return;
   760→    }
   761→    // AI decides: accepts only if clearly worse and past the opening
   762→    const evalCp = engineInfo?.evalCp ?? 0;
   763→    const aiIsWhite = engineColor === "w";
   764→    const aiEval = aiIsWhite ? evalCp : -evalCp;
   765→    if (aiEval < -180 && history.length >= 24) {
   766→      endGame({ winner: null, reason: "agreement" });
   767→      toast("Draw agreed", { description: "The Grandmaster accepts your offer." });
   768→    } else {
   769→      toast("Draw declined", {
   770→        description: "The Grandmaster smells blood and plays on.",
   771→      });
   772→    }
   773→  }, [status, mode, turn, engineInfo, engineColor, history.length, endGame]);
   774→
   775→  const resolveDrawOffer = useCallback(
   776→    (accept: boolean) => {
   777→      setDrawOfferPending(null);
   778→      if (accept) endGame({ winner: null, reason: "agreement" });
   779→    },
   780→    [endGame]
   781→  );
   782→
   783→  const requestHint = useCallback(async () => {
   784→    if (status !== "active" || thinking || !isHumanTurn) return;
   785→    if (hintsUsed >= 3) {
   786→      toast("No hints left", {
   787→        description: "Three whispers per game — the rest is destiny.",
   788→      });
   789→      return;
   790→    }
   791→    setHintsUsed((h) => h + 1);
   792→    setThinking(true);
   793→    try {
   794→      const res = await fetch("/api/engine", {
   795→        method: "POST",
   796→        headers: { "Content-Type": "application/json" },
   797→        body: JSON.stringify({ fen: gameRef.current.fen(), level: "hard" }),
   798→      });
   799→      if (!res.ok) throw new Error();
   800→      const data = (await res.json()) as { from: Square; to: Square };
   801→      setHint({ from: data.from, to: data.to });
   802→      setSelected(null);
   803→      setLegalForSelected([]);
   804→      window.setTimeout(() => setHint((h) => (h?.from === data.from && h?.to === data.to ? null : h)), 2600);
   805→    } catch {
   806→      toast.error("Hint unavailable right now.");
   807→    } finally {
   808→      setThinking(false);
   809→    }
   810→  }, [status, thinking, isHumanTurn, hintsUsed]);
   811→
   812→  const flipBoard = useCallback(() => {
   813→    setOrientation((o) => (o === "white" ? "black" : "white"));
   814→  }, []);
   815→
   816→  const goToHistory = useCallback((index: number | null) => {
   817→    setViewIndex(index);
   818→    setSelected(null);
   819→    setLegalForSelected([]);
   820→  }, []);
   821→
   822→  const primeAudioOnGesture = useCallback(() => {
   823→    primeAudio();
   824→  }, []);
   825→
   826→  return {
   827→    // state
   828→    fen,
   829→    pieces: displayPieces,
   830→    livePieces: pieces,
   831→    history,
   832→    turn,
   833→    lastMove,
   834→    inCheck,
   835→    checkedSquare,
   836→    selected,
   837→    legalForSelected,
   838→    pendingPromotion,
   839→    status,
   840→    result,
   841→    orientation,
   842→    humanColor,
   843→    engineColor: sideOf(engineColor),
   844→    mode,
   845→    difficulty,
   846→    soundOn,
   847→    showLegal,
   848→    showCoords,
   849→    autoQueen,
   850→    preferredSide,
   851→    clocks,
   852→    activeTimeControl,
   853→    thinking,
   854→    engineInfo,
   855→    hint,
   856→    hintsUsed,
   857→    viewIndex,
   858→    viewing,
   859→    interactive,
   860→    started,
   861→    stats,
   862→    drawOfferPending,
   863→    capturedBy,
   864→    // actions
   865→    selectSquare,
   866→    attemptMove,
   867→    promote,
   868→    cancelPromotion,
   869→    newGame,
   870→    undo,
   871→    resign,
   872→    offerDraw,
   873→    resolveDrawOffer,
   874→    requestHint,
   875→    flipBoard,
   876→    goToHistory,
   877→    setSoundOn,
   878→    setShowLegal,
   879→    setShowCoords,
   880→    setAutoQueen,
   881→    setMode,
   882→    setDifficulty,
   883→    setPreferredSide,
   884→    primeAudioOnGesture,
   885→  };
   886→}
   887→
   888→export type ChessGame = ReturnType<typeof useChessGame>;
