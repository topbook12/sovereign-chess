    85→    85→function loadJSON<T>(key: string, fallback: T): T {
    86→    86→  if (typeof window === "undefined") return fallback;
    87→    87→  try {
    88→    88→    const raw = window.localStorage.getItem(key);
    89→    89→    return raw ? { ...fallback, ...(JSON.parse(raw) as object) } as T : fallback;
    90→    90→  } catch {
    91→    91→    return fallback;
    92→    92→  }
    93→    93→}

   102→   102→function fenToPieces(fen: string, opts?: { spawn?: boolean }): PieceView[] {
   103→   103→  const placement = fen.split(" ")[0];
   104→   104→  const rows = placement.split("/");
   105→   105→  const out: PieceView[] = [];
   106→   106→  let n = 0;
   107→   107→  for (let r = 0; r < 8; r++) {
   108→   108→    let file = 0;
   109→   109→    for (const ch of rows[r]) {
   110→   110→      if (/\d/.test(ch)) {
   111→   111→        file += Number(ch);
   112→   112→        continue;
   113→   113→      }
   114→   114→      const color: Color = ch === ch.toUpperCase() ? "w" : "b";
   115→   115→      const type = ch.toLowerCase() as PieceSymbol;
   116→   116→      const rank = 8 - r;
   117→   117→      const square = `${String.fromCharCode(97 + file)}${rank}` as Square;
   118→   118→      out.push({
   119→   119→        id: `p${n}-${color}${type}-${square}`,
   120→   120→        square,
   121→   121→        type,
   122→   122→        color,
   123→   123→        spawn: opts?.spawn,
   124→   124→        spawnDelay: opts?.spawn ? n * 14 : 0,
   125→   125→      });
   126→   126→      n++;
   127→   127→      file++;
   128→   128→    }
   129→   129→  }
   130→   130→  return out;
   131→   131→}

   137→   137→  return "fifty";

   457→   457→  const applyMove = useCallback(
   458→   458→    (mv: { from: Square; to: Square; promotion?: PromotionPiece }): boolean => {
   459→   459→      const g = gameRef.current;
   460→   460→      let move: Move;
   461→   461→      try {
   462→   462→        move = g.move({ from: mv.from, to: mv.to, promotion: mv.promotion ?? "q" });
   463→   463→      } catch {
   464→   464→        playSound("illegal");
   465→   465→        return false;
   466→   466→      }
   467→   467→
   468→   468→      idRef.current++;
   469→   469→      // reconcile animated pieces
   470→   470→      setPieces((prev) => {
   471→   471→        const next: PieceView[] = [];
   472→   472→        const epSquare = move.flags.includes("e")
   473→   473→          ? (`${move.to[0]}${move.from[1]}` as Square)
   474→   474→          : null;
   475→   475→        for (const p of prev) {
   476→   476→          if (p.captured) continue; // already fading
   477→   477→          if (epSquare && p.square === epSquare && p.color !== move.color) {
   478→   478→            next.push({ ...p, id: `${p.id}-x${idRef.current}`, captured: true });
   479→   479→            continue;
   480→   480→          }
   481→   481→          if (p.square === move.to && p.color !== move.color) {
   482→   482→            next.push({ ...p, id: `${p.id}-x${idRef.current}`, captured: true });
   483→   483→            continue;
   484→   484→          }
   485→   485→          if (p.square === move.from && p.color === move.color && p.type === move.piece) {
   486→   486→            next.push({
   487→   487→              ...p,
   488→   488→              square: move.to,
   489→   489→              type: move.promotion ?? p.type,
   490→   490→            });
   491→   491→            continue;
   492→   492→          }
   493→   493→          // castling rook
   494→   494→          if (
   495→   495→            move.piece === "k" &&
   496→   496→            p.type === "r" &&
   497→   497→            p.color === move.color &&
   498→   498→            (move.flags.includes("k") &&
   499→   499→              p.square === (move.color === "w" ? "h1" : "h8"))
   500→   500→          ) {
   501→   501→            next.push({
   502→   502→              ...p,
   503→   503→              square: (move.color === "w" ? "f1" : "f8") as Square,
   504→   504→            });
   505→   505→            continue;
   506→   506→          }
   507→   507→          if (
   508→   508→            move.piece === "k" &&
   509→   509→            p.type === "r" &&
   510→   510→            p.color === move.color &&
   511→   511→            (move.flags.includes("q") &&
   512→   512→              p.square === (move.color === "w" ? "a1" : "a8"))
   513→   513→          ) {
   514→   514→            next.push({
   515→   515→              ...p,
   516→   516→              square: (move.color === "w" ? "d1" : "d8") as Square,
   517→   517→            });
   518→   518→            continue;
   519→   519→          }
   520→   520→          next.push(p);
   521→   521→        }
   522→   522→        return next;
   523→   523→      });
   524→   524→
   525→   525→      // cleanup captured fade-out pieces
   526→   526→      window.setTimeout(() => {
   527→   527→        setPieces((prev) => prev.filter((p) => !p.captured));
   528→   528→      }, 320);
   529→   529→
   530→   530→      refreshBoardState(false);
   531→   531→
   532→   532→      // sounds
   533→   533→      if (move.flags.includes("k") || move.flags.includes("q")) playSound("castle");
   534→   534→      else if (move.captured) playSound("capture");
   535→   535→      else if (move.promotion) playSound("promote");
   536→   536→      else playSound("move");
   537→   537→
   538→   538→      const gg = gameRef.current;
   539→   539→      if (gg.isCheckmate()) {
   540→   540→        window.setTimeout(
   541→   541→          () =>
   542→   542→            endGame({
   543→   543→              winner: sideOf(gg.turn() === "w" ? "b" : "w"),
   544→   544→              reason: "checkmate",
   545→   545→            }),
   546→   546→          420
   547→   547→        );
   548→   548→      } else if (gg.isDraw()) {
   549→   549→        window.setTimeout(() => endGame({ winner: null, reason: classifyDraw(gg) }), 420);
   550→   550→      } else if (gg.isCheck()) {
   551→   551→        window.setTimeout(() => playSound("check"), 90);
   552→   552→      }
   553→   553→
   554→   554→      setSelected(null);
   555→   555→      setLegalForSelected([]);
   556→   556→      setPendingPromotion(null);
   557→   557→      return true;
   558→   558→    },
   559→   559→    [endGame, refreshBoardState]
   560→   560→  );
   561→   561→
   562→   562→  useEffect(() => {
   563→   563→    applyMoveRef.current = applyMove;
   564→   564→  }, [applyMove]);
   565→   565→
   566→   566→  const attemptMove = useCallback(
   567→   567→    (from: Square, to: Square): boolean => {
   568→   568→      if (!interactive) return false;
   569→   569→      const g = gameRef.current;
   570→   570→      const candidates = g
   571→   571→        .moves({ square: from, verbose: true })
   572→   572→        .filter((m) => m.to === to);
   573→   573→      if (candidates.length === 0) {
   574→   574→        playSound("illegal");
   575→   575→        return false;
   576→   576→      }
   577→   577→      if (candidates[0].promotion) {
   578→   578→        if (autoQueen) {
   579→   579→          return applyMove({ from, to, promotion: "q" });
   580→   580→        }
   581→   581→        setPendingPromotion({ from, to });
   582→   582→        return true;
   583→   583→      }
   584→   584→      return applyMove({ from, to });
   585→   585→    },
   586→   586→    [interactive, autoQueen, applyMove]
   587→   587→  );
   588→   588→
   589→   589→  const selectSquare = useCallback(
   590→   590→    (square: Square | null) => {
   591→   591→      if (!interactive) {
   592→   592→        setSelected(null);
   593→   593→        setLegalForSelected([]);
   594→   594→        return;
   595→   595→      }
   596→   596→      if (square === null) {
   597→   597→        setSelected(null);
   598→   598→        setLegalForSelected([]);
   599→   599→        return;
   600→   600→      }
   601→   601→      if (selected === square) {
   602→   602→        setSelected(null);
   603→   603→        setLegalForSelected([]);
   604→   604→        return;
   605→   605→      }
   606→   606→      const g = gameRef.current;
   607→   607→      const piece = g.get(square);
   608→   608→      if (piece && piece.color === g.turn()) {
   609→   609→        setSelected(square);
   610→   610→        setLegalForSelected(g.moves({ square, verbose: true }));
   611→   611→        playSound("select");
   612→   612→        return;
   613→   613→      }
   614→   614→      // clicking a target square
   615→   615→      if (selected && legalForSelected.some((m) => m.to === square)) {
   616→   616→        attemptMove(selected, square);
   617→   617→        return;
   618→   618→      }
   619→   619→      setSelected(null);
   620→   620→      setLegalForSelected([]);
   621→   621→    },
   622→   622→    [interactive, selected, legalForSelected, attemptMove]
   623→   623→  );
   624→   624→
   625→   625→  const promote = useCallback(
   626→   626→    (piece: PromotionPiece) => {
   627→   627→      if (!pendingPromotion) return;
   628→   628→      applyMove({ from: pendingPromotion.from, to: pendingPromotion.to, promotion: piece });
   629→   629→    },
   630→   630→    [pendingPromotion, applyMove]
   631→   631→  );
   632→   632→
   633→   633→  const cancelPromotion = useCallback(() => {
   634→   634→    setPendingPromotion(null);
   635→   635→    setSelected(null);
   636→   636→    setLegalForSelected([]);
   637→   637→  }, []);
   638→   638→
   639→   639→  // ---------- engine auto-play effect ----------
   640→   640→  const engineTurnKey = `${fen}|${mode}|${status}|${viewIndex}|${difficulty}|${pendingPromotion ? 1 : 0}|${engineRetryTick}`;
   641→   641→  const lastEngineTurnKey = useRef("");
   642→   642→  useEffect(() => {
   643→   643→    if (mode !== "ai" || status !== "active" || viewing || pendingPromotion) {
   644→   644→      return;
   645→   645→    }
   646→   646→    if (gameRef.current.turn() !== engineColor) return;
   647→   647→    if (lastEngineTurnKey.current === engineTurnKey) return;
   648→   648→    const reqId = ++engineReqRef.current;
   649→   649→    // Latch only when the request actually fires — if the user scrubs history
   650→   650→    // during the scheduling window the cleanup below cancels the timer, the
   651→   651→    // latch stays unset, and returning to Live reschedules cleanly.
   652→   652→    const t = window.setTimeout(() => {
   653→   653→      lastEngineTurnKey.current = engineTurnKey;
   654→   654→      void requestEngineMove(reqId);
   655→   655→    }, 240);
   656→   656→    return () => window.clearTimeout(t);
   657→   657→  }, [engineTurnKey]);
   658→   658→
   659→   659→  // ---------- clock ----------
   660→   660→  useEffect(() => {
   661→   661→    clockSideRef.current = turn;
   662→   662→  }, [turn]);
   663→   663→
   664→   664→  useEffect(() => {
   665→   665→    if (status !== "active" || activeTimeControl.initialMs === 0) return;
   666→   666→    if (history.length === 0) return; // clock starts after the first move
   667→   667→    let last = Date.now();
   668→   668→    const iv = window.setInterval(() => {
   669→   669→      const now = Date.now();
   670→   670→      const dt = now - last;
   671→   671→      last = now;
   672→   672→      setClocks((prev) => {
   673→   673→        const side = clockSideRef.current;
   674→   674→        return { ...prev, [side]: Math.max(0, prev[side] - dt) };
   675→   675→      });
   676→   676→    }, 100);
   677→   677→    return () => window.clearInterval(iv);
   678→   678→  }, [status, activeTimeControl.initialMs, turn, fen]);
   679→   679→
   680→   680→  useEffect(() => {
   681→   681→    if (status !== "active" || activeTimeControl.initialMs === 0) return;
   682→   682→    if (clocks.w <= 0) endGame({ winner: "black", reason: "timeout" });
   683→   683→    else if (clocks.b <= 0) endGame({ winner: "white", reason: "timeout" });
   684→   684→  }, [clocks, status, activeTimeControl.initialMs, endGame]);
   685→   685→
   686→   686→  // ---------- actions ----------
   687→   687→  const newGame = useCallback(
   688→   688→    (opts?: {
   689→   689→      mode?: GameMode;
   690→   690→      difficulty?: Difficulty;
   691→   691→      side?: Side | "random";
   692→   692→      timeControlId?: string;
   693→   693→    }) => {
   694→   694→      engineReqRef.current++;
   695→   695→      const nextMode = opts?.mode ?? mode;
   696→   696→      const nextSide: Side =
   697→   697→        opts?.side === "random"
   698→   698→          ? Math.random() < 0.5
   699→   699→            ? "white"
   700→   700→            : "black"
   701→   701→          : (opts?.side ?? preferredSide);
   702→   702→      const tc =
   703→   703→        TIME_CONTROLS.find((t) => t.id === opts?.timeControlId) ??
   704→   704→        activeTimeControl;
   705→   705→
   706→   706→      gameRef.current = new Chess();
   707→   707→      idRef.current++;
   708→   708→      endedRef.current = false;
   709→   709→      lastEngineTurnKey.current = "";
   710→   710→      setMode(nextMode);
   711→   711→      if (opts?.difficulty) setDifficulty(opts.difficulty);
   712→   712→      if (opts?.side) setPreferredSide(nextSide === "white" ? "white" : "black");
   713→   713→      setActiveTimeControl(tc);
   714→   714→      setHumanColor(nextSide);
   715→   715→      setOrientation(nextSide);
   716→   716→      setFen(gameRef.current.fen());
   717→   717→      setPieces(fenToPieces(gameRef.current.fen(), { spawn: true }));
   718→   718→      setHistory([]);
   719→   719→      setTurn("w");
   720→   720→      setLastMove(null);
   721→   721→      setInCheck(false);
   722→   722→      setCheckedSquare(null);
   723→   723→      setSelected(null);
   724→   724→      setLegalForSelected([]);
   725→   725→      setPendingPromotion(null);
   726→   726→      setStatus("active");
   727→   727→      setResult(null);
   728→   728→      setDrawOfferPending(null);
   729→   729→      setClocks({ w: tc.initialMs, b: tc.initialMs });
   730→   730→      setThinking(false);
   731→   731→      setEngineInfo(null);
   732→   732→      setHint(null);
   733→   733→      setHintsUsed(0);
   734→   734→      setViewIndex(null);
   735→   735→    },
   736→   736→    [mode, preferredSide, activeTimeControl]
   737→   737→  );
   738→   738→
   739→   739→  const undo = useCallback(() => {
   740→   740→    engineReqRef.current++;
   741→   741→    setThinking(false);
   742→   742→    const g = gameRef.current;
   743→   743→    if (g.history().length === 0) return;
   744→   744→    if (mode === "ai" && g.turn() === charOf(humanColor) && g.history().length >= 2) {
   745→   745→      g.undo(); // engine reply
   746→   746→      g.undo(); // human move
   747→   747→    } else {
   748→   748→      g.undo();
   749→   749→    }
   750→   750→    setSelected(null);
   751→   751→    setLegalForSelected([]);
   752→   752→    setPendingPromotion(null);
   753→   753→    setHint(null);
   754→   754→    setViewIndex(null);
   755→   755→    endedRef.current = false;
   756→   756→    if (status === "ended") {
   757→   757→      setStatus("active");
   758→   758→      setResult(null);
   759→   759→    }
   760→   760→    lastEngineTurnKey.current = "";
   761→   761→    refreshBoardState();
   762→   762→    playSound("move");
   763→   763→  }, [mode, humanColor, status, refreshBoardState]);
   764→   764→
   765→   765→  const resign = useCallback(() => {
   766→   766→    if (status !== "active") return;
   767→   767→    const loser: Side = mode === "ai" ? humanColor : sideOf(turn);
   768→   768→    endGame({ winner: loser === "white" ? "black" : "white", reason: "resignation" });
   769→   769→    toast(`${loser === "white" ? "White" : "Black"} resigned.`, {
   770→   770→      description: "The crown changes hands.",
   771→   771→    });
   772→   772→  }, [status, mode, humanColor, turn, endGame]);
   773→   773→
   774→   774→  const offerDraw = useCallback(() => {
   775→   775→    if (status !== "active") return;
   776→   776→    if (mode === "human") {
   777→   777→      setDrawOfferPending(turn);
   778→   778→      return;
   779→   779→    }
   780→   780→    // AI decides: accepts only if clearly worse and past the opening
   781→   781→    const evalCp = engineInfo?.evalCp ?? 0;
   782→   782→    const aiIsWhite = engineColor === "w";
   783→   783→    const aiEval = aiIsWhite ? evalCp : -evalCp;
   784→   784→    if (aiEval < -180 && history.length >= 24) {
   785→   785→      endGame({ winner: null, reason: "agreement" });
   786→   786→      toast("Draw agreed", { description: "The Grandmaster accepts your offer." });
   787→   787→    } else {
   788→   788→      toast("Draw declined", {
   789→   789→        description: "The Grandmaster smells blood and plays on.",
   790→   790→      });
   791→   791→    }
   792→   792→  }, [status, mode, turn, engineInfo, engineColor, history.length, endGame]);
   793→   793→
   794→   794→  const resolveDrawOffer = useCallback(
   795→   795→    (accept: boolean) => {
   796→   796→      setDrawOfferPending(null);
   797→   797→      if (accept) endGame({ winner: null, reason: "agreement" });
   798→   798→    },
   799→   799→    [endGame]
   800→   800→  );
   801→   801→
   802→   802→  const requestHint = useCallback(async () => {
   803→   803→    if (status !== "active" || thinking || !isHumanTurn) return;
   804→   804→    if (hintsUsed >= 3) {
   805→   805→      toast("No hints left", {
   806→   806→        description: "Three whispers per game — the rest is destiny.",
   807→   807→      });
   808→   808→      return;
   809→   809→    }
   810→   810→    setHintsUsed((h) => h + 1);
   811→   811→    setThinking(true);
   812→   812→    try {
   813→   813→      const res = await fetch("/api/engine", {
   814→   814→        method: "POST",
   815→   815→        headers: { "Content-Type": "application/json" },
   816→   816→        body: JSON.stringify({ fen: gameRef.current.fen(), level: "hard" }),
   817→   817→      });
   818→   818→      if (!res.ok) throw new Error();
   819→   819→      const data = (await res.json()) as { from: Square; to: Square };
   820→   820→      setHint({ from: data.from, to: data.to });
   821→   821→      setSelected(null);
   822→   822→      setLegalForSelected([]);
   823→   823→      window.setTimeout(() => setHint((h) => (h?.from === data.from && h?.to === data.to ? null : h)), 2600);
   824→   824→    } catch {
   825→   825→      toast.error("Hint unavailable right now.");
   826→   826→    } finally {
   827→   827→      setThinking(false);
   828→   828→    }
   829→   829→  }, [status, thinking, isHumanTurn, hintsUsed]);
   830→   830→
   831→   831→  const flipBoard = useCallback(() => {
   832→   832→    setOrientation((o) => (o === "white" ? "black" : "white"));
   833→   833→  }, []);

   836→   836→    setViewIndex(index);
   837→   837→    setSelected(null);
   838→   838→    setLegalForSelected([]);
   839→   839→  }, []);
   840→   840→
   841→   841→  const primeAudioOnGesture = useCallback(() => {
   842→   842→    primeAudio();
   843→   843→  }, []);

   858→   858→    status,
   859→   859→    result,
   860→   860→    orientation,
   861→   861→    humanColor,
   862→   862→    engineColor: sideOf(engineColor),
   863→   863→    mode,
   864→   864→    difficulty,
   865→   865→    soundOn,
   866→   866→    showLegal,
   867→   867→    showCoords,
   868→   868→    autoQueen,
   869→   869→    preferredSide,
