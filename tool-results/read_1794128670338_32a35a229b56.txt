    85→function loadJSON<T>(key: string, fallback: T): T {
    86→  if (typeof window === "undefined") return fallback;
    87→  try {
    88→    const raw = window.localStorage.getItem(key);
    89→    return raw ? { ...fallback, ...(JSON.parse(raw) as object) } as T : fallback;
    90→  } catch {
    91→    return fallback;
    92→  }
    93→}

    95→function sideOf(c: Color): Side {
    96→  return c === "w" ? "white" : "black";

    99→  return s === "white" ? "w" : "b";

   102→function fenToPieces(fen: string, opts?: { spawn?: boolean }): PieceView[] {
   103→  const placement = fen.split(" ")[0];
   104→  const rows = placement.split("/");
   105→  const out: PieceView[] = [];
   106→  let n = 0;
   107→  for (let r = 0; r < 8; r++) {
   108→    let file = 0;
   109→    for (const ch of rows[r]) {
   110→      if (/\d/.test(ch)) {
   111→        file += Number(ch);
   112→        continue;
   113→      }
   114→      const color: Color = ch === ch.toUpperCase() ? "w" : "b";
   115→      const type = ch.toLowerCase() as PieceSymbol;
   116→      const rank = 8 - r;
   117→      const square = `${String.fromCharCode(97 + file)}${rank}` as Square;
   118→      out.push({
   119→        id: `p${n}-${color}${type}-${square}`,
   120→        square,
   121→        type,
   122→        color,
   123→        spawn: opts?.spawn,
   124→        spawnDelay: opts?.spawn ? n * 14 : 0,
   125→      });
   126→      n++;
   127→      file++;
   128→    }
   129→  }
   130→  return out;
   131→}

   133→function classifyDraw(game: Chess): GameResult["reason"] {
   134→  if (game.isStalemate()) return "stalemate";
   135→  if (game.isThreefoldRepetition()) return "threefold";
   136→  if (game.isInsufficientMaterial()) return "insufficient";
   137→  return "fifty";
   138→}

   373→        else if (r.winner === humanColor) playSound("win");
   374→        else playSound("lose");
   375→      }
   376→      if (opts?.countStats !== false && mode === "ai") bumpStats(r);
   377→    },
   378→    [mode, humanColor, bumpStats]

   380→

   382→    from: Square;

   387→  // ---------- engine ----------
   388→  const requestEngineMove = useCallback(
   389→    async (reqId: number) => {
   390→      const g = gameRef.current;
   391→      setThinking(true);
   392→      const t0 = Date.now();
   393→      try {
   394→        const res = await fetch("/api/engine", {
   395→          method: "POST",
   396→          headers: { "Content-Type": "application/json" },
   397→          body: JSON.stringify({ fen: g.fen(), level: difficulty }),
   398→        });
   399→        if (!res.ok) throw new Error(`Engine error ${res.status}`);
   400→        const data = (await res.json()) as {
   401→          from: Square;
   402→          to: Square;
   403→          promotion?: string;
   404→          san: string;
   405→          evalCp: number;
   406→          mateIn?: number;
   407→          depth: number;
   408→          nodes: number;
   409→          timeMs: number;
   410→        };
   411→        if (
   412→          reqId !== engineReqRef.current ||
   413→          status !== "active" ||
   414→          gameRef.current.fen() !== g.fen()
   415→        ) {
   416→          return; // stale
   417→        }
   418→        engineFailRef.current = 0;
   419→        // reveal the move with a brief arrow, then play it
   420→        setHint({ from: data.from, to: data.to });
   421→        const wait = Math.max(180, 600 - (Date.now() - t0));
   422→        await new Promise((r) => setTimeout(r, wait));
   423→        if (reqId !== engineReqRef.current || status !== "active") return;
   424→        setHint(null);
   425→        applyMoveRef.current({
   426→          from: data.from,
   427→          to: data.to,
   428→          promotion: (data.promotion as PromotionPiece) ?? "q",
   429→        });
   430→        setEngineInfo({
   431→          evalCp: data.evalCp,
   432→          mateIn: data.mateIn,
   433→          depth: data.depth,
   434→          nodes: data.nodes,
   435→          timeMs: data.timeMs,
   436→          level: difficulty,
   437→        });
   438→      } catch {
   439→        if (reqId === engineReqRef.current) {
   440→          setThinking(false);
   441→          lastEngineTurnKey.current = "";
   442→          if (engineFailRef.current === 0) {
   443→            toast.error("Engine hiccup — retrying…");
   444→          }
   445→          engineFailRef.current++;
   446→          // automatic retry with a small backoff keeps the game alive
   447→          window.setTimeout(() => setEngineRetryTick((t) => t + 1), 1500);
   448→        }
   449→      } finally {
   450→        if (reqId === engineReqRef.current) setThinking(false);
   451→      }
   452→    },
   453→    [difficulty, status]
   454→  );
   455→
   456→  // ---------- moving ----------
   457→  const applyMove = useCallback(
   458→    (mv: { from: Square; to: Square; promotion?: PromotionPiece }): boolean => {
   459→      const g = gameRef.current;
   460→      let move: Move;
   461→      try {
   462→        move = g.move({ from: mv.from, to: mv.to, promotion: mv.promotion ?? "q" });
   463→      } catch {
   464→        playSound("illegal");
   465→        return false;
   466→      }
   467→
   468→      idRef.current++;
   469→      // reconcile animated pieces
   470→      setPieces((prev) => {
   471→        const next: PieceView[] = [];
   472→        const epSquare = move.flags.includes("e")
   473→          ? (`${move.to[0]}${move.from[1]}` as Square)
   474→          : null;
   475→        for (const p of prev) {
   476→          if (p.captured) continue; // already fading
   477→          if (epSquare && p.square === epSquare && p.color !== move.color) {
   478→            next.push({ ...p, id: `${p.id}-x${idRef.current}`, captured: true });
   479→            continue;
   480→          }
   481→          if (p.square === move.to && p.color !== move.color) {
   482→            next.push({ ...p, id: `${p.id}-x${idRef.current}`, captured: true });
   483→            continue;
   484→          }
   485→          if (p.square === move.from && p.color === move.color && p.type === move.piece) {
   486→            next.push({
   487→              ...p,
   488→              square: move.to,
   489→              type: move.promotion ?? p.type,
   490→            });
   491→            continue;
   492→          }
   493→          // castling rook
   494→          if (
   495→            move.piece === "k" &&
   496→            p.type === "r" &&
   497→            p.color === move.color &&
   498→            (move.flags.includes("k") &&
   499→              p.square === (move.color === "w" ? "h1" : "h8"))
   500→          ) {
   501→            next.push({
   502→              ...p,
   503→              square: (move.color === "w" ? "f1" : "f8") as Square,
   504→            });
   505→            continue;
   506→          }
   507→          if (
   508→            move.piece === "k" &&
   509→            p.type === "r" &&
   510→            p.color === move.color &&
   511→            (move.flags.includes("q") &&
   512→              p.square === (move.color === "w" ? "a1" : "a8"))
   513→          ) {
   514→            next.push({
   515→              ...p,
   516→              square: (move.color === "w" ? "d1" : "d8") as Square,
   517→            });
   518→            continue;
   519→          }
   520→          next.push(p);
   521→        }
   522→        return next;
   523→      });
   524→
   525→      // cleanup captured fade-out pieces
   526→      window.setTimeout(() => {
   527→        setPieces((prev) => prev.filter((p) => !p.captured));
   528→      }, 320);
   529→
   530→      refreshBoardState(false);
   531→
   532→      // sounds
   533→      if (move.flags.includes("k") || move.flags.includes("q")) playSound("castle");
   534→      else if (move.captured) playSound("capture");
   535→      else if (move.promotion) playSound("promote");
   536→      else playSound("move");
   537→
   538→      const gg = gameRef.current;
   539→      if (gg.isCheckmate()) {
   540→        window.setTimeout(
   541→          () =>
   542→            endGame({
   543→              winner: sideOf(gg.turn() === "w" ? "b" : "w"),
   544→              reason: "checkmate",
   545→            }),
   546→          420
   547→        );
   548→      } else if (gg.isDraw()) {
   549→        window.setTimeout(() => endGame({ winner: null, reason: classifyDraw(gg) }), 420);
   550→      } else if (gg.isCheck()) {
   551→        window.setTimeout(() => playSound("check"), 90);
   552→      }
   553→
   554→      setSelected(null);
   555→      setLegalForSelected([]);
   556→      setPendingPromotion(null);
   557→      return true;
   558→    },
   559→    [endGame, refreshBoardState]
   560→  );
   561→
   562→  useEffect(() => {
   563→    applyMoveRef.current = applyMove;
   564→  }, [applyMove]);
   565→
   566→  const attemptMove = useCallback(
   567→    (from: Square, to: Square): boolean => {
   568→      if (!interactive) return false;
   569→      const g = gameRef.current;
   570→      const candidates = g
   571→        .moves({ square: from, verbose: true })
   572→        .filter((m) => m.to === to);
   573→      if (candidates.length === 0) {
   574→        playSound("illegal");
   575→        return false;
   576→      }
   577→      if (candidates[0].promotion) {
   578→        if (autoQueen) {
   579→          return applyMove({ from, to, promotion: "q" });
   580→        }
   581→        setPendingPromotion({ from, to });
   582→        return true;
   583→      }
   584→      return applyMove({ from, to });
   585→    },
   586→    [interactive, autoQueen, applyMove]
   587→  );
   588→
   589→  const selectSquare = useCallback(
   590→    (square: Square | null) => {
   591→      if (!interactive) {
   592→        setSelected(null);
   593→        setLegalForSelected([]);
   594→        return;
   595→      }
   596→      if (square === null) {
   597→        setSelected(null);
   598→        setLegalForSelected([]);
   599→        return;
   600→      }
   601→      if (selected === square) {
   602→        setSelected(null);
   603→        setLegalForSelected([]);
   604→        return;
   605→      }
   606→      const g = gameRef.current;
   607→      const piece = g.get(square);
   608→      if (piece && piece.color === g.turn()) {
   609→        setSelected(square);
   610→        setLegalForSelected(g.moves({ square, verbose: true }));
   611→        playSound("select");
   612→        return;
   613→      }
   614→      // clicking a target square
   615→      if (selected && legalForSelected.some((m) => m.to === square)) {
   616→        attemptMove(selected, square);
   617→        return;
   618→      }
   619→      setSelected(null);
   620→      setLegalForSelected([]);
   621→    },
   622→    [interactive, selected, legalForSelected, attemptMove]
   623→  );
   624→
   625→  const promote = useCallback(
   626→    (piece: PromotionPiece) => {
   627→      if (!pendingPromotion) return;
   628→      applyMove({ from: pendingPromotion.from, to: pendingPromotion.to, promotion: piece });
   629→    },
   630→    [pendingPromotion, applyMove]
   631→  );
   632→
   633→  const cancelPromotion = useCallback(() => {
   634→    setPendingPromotion(null);
   635→    setSelected(null);
   636→    setLegalForSelected([]);
   637→  }, []);
   638→
   639→  // ---------- engine auto-play effect ----------
   640→  const engineTurnKey = `${fen}|${mode}|${status}|${viewIndex}|${difficulty}|${pendingPromotion ? 1 : 0}|${engineRetryTick}`;
   641→  const lastEngineTurnKey = useRef("");
   642→  useEffect(() => {
   643→    if (mode !== "ai" || status !== "active" || viewing || pendingPromotion) {
   644→      return;
   645→    }
   646→    if (gameRef.current.turn() !== engineColor) return;
   647→    if (lastEngineTurnKey.current === engineTurnKey) return;
   648→    const reqId = ++engineReqRef.current;
   649→    // Latch only when the request actually fires — if the user scrubs history
   650→    // during the scheduling window the cleanup below cancels the timer, the
   651→    // latch stays unset, and returning to Live reschedules cleanly.
   652→    const t = window.setTimeout(() => {
   653→      lastEngineTurnKey.current = engineTurnKey;
   654→      void requestEngineMove(reqId);
   655→    }, 240);
   656→    return () => window.clearTimeout(t);
   657→  }, [engineTurnKey]);
   658→
   659→  // ---------- clock ----------
   660→  useEffect(() => {
   661→    clockSideRef.current = turn;
   662→  }, [turn]);
   663→
   664→  useEffect(() => {
   665→    if (status !== "active" || activeTimeControl.initialMs === 0) return;
   666→    if (history.length === 0) return; // clock starts after the first move
   667→    let last = Date.now();
   668→    const iv = window.setInterval(() => {
   669→      const now = Date.now();
   670→      const dt = now - last;
   671→      last = now;
   672→      setClocks((prev) => {
   673→        const side = clockSideRef.current;
   674→        return { ...prev, [side]: Math.max(0, prev[side] - dt) };
   675→      });
   676→    }, 100);
   677→    return () => window.clearInterval(iv);
   678→  }, [status, activeTimeControl.initialMs, turn, fen]);
   679→
   680→  useEffect(() => {
   681→    if (status !== "active" || activeTimeControl.initialMs === 0) return;
   682→    if (clocks.w <= 0) endGame({ winner: "black", reason: "timeout" });
   683→    else if (clocks.b <= 0) endGame({ winner: "white", reason: "timeout" });
   684→  }, [clocks, status, activeTimeControl.initialMs, endGame]);
   685→
   686→  // ---------- actions ----------
   687→  const newGame = useCallback(
   688→    (opts?: {
   689→      mode?: GameMode;
   690→      difficulty?: Difficulty;
   691→      side?: Side | "random";
   692→      timeControlId?: string;
   693→    }) => {
   694→      engineReqRef.current++;
   695→      const nextMode = opts?.mode ?? mode;
   696→      const nextSide: Side =
   697→        opts?.side === "random"
   698→          ? Math.random() < 0.5
   699→            ? "white"
   700→            : "black"
   701→          : (opts?.side ?? preferredSide);
   702→      const tc =
   703→        TIME_CONTROLS.find((t) => t.id === opts?.timeControlId) ??
   704→        activeTimeControl;
   705→
   706→      gameRef.current = new Chess();
   707→      idRef.current++;
   708→      endedRef.current = false;
   709→      lastEngineTurnKey.current = "";
   710→      setMode(nextMode);
   711→      if (opts?.difficulty) setDifficulty(opts.difficulty);
   712→      if (opts?.side) setPreferredSide(nextSide === "white" ? "white" : "black");
   713→      setActiveTimeControl(tc);
   714→      setHumanColor(nextSide);
   715→      setOrientation(nextSide);
   716→      setFen(gameRef.current.fen());
   717→      setPieces(fenToPieces(gameRef.current.fen(), { spawn: true }));
   718→      setHistory([]);
   719→      setTurn("w");
   720→      setLastMove(null);
   721→      setInCheck(false);
   722→      setCheckedSquare(null);
   723→      setSelected(null);
   724→      setLegalForSelected([]);
   725→      setPendingPromotion(null);
   726→      setStatus("active");
   727→      setResult(null);
   728→      setDrawOfferPending(null);
   729→      setClocks({ w: tc.initialMs, b: tc.initialMs });
   730→      setThinking(false);
   731→      setEngineInfo(null);
   732→      setHint(null);
   733→      setHintsUsed(0);
   734→      setViewIndex(null);
   735→    },
   736→    [mode, preferredSide, activeTimeControl]
   737→  );
   738→
   739→  const undo = useCallback(() => {
   740→    engineReqRef.current++;
   741→    setThinking(false);
   742→    const g = gameRef.current;
   743→    if (g.history().length === 0) return;
   744→    if (mode === "ai" && g.turn() === charOf(humanColor) && g.history().length >= 2) {
   745→      g.undo(); // engine reply
   746→      g.undo(); // human move
   747→    } else {
   748→      g.undo();
   749→    }
   750→    setSelected(null);
   751→    setLegalForSelected([]);
   752→    setPendingPromotion(null);
   753→    setHint(null);
   754→    setViewIndex(null);
   755→    endedRef.current = false;
   756→    if (status === "ended") {
   757→      setStatus("active");
   758→      setResult(null);
   759→    }
   760→    lastEngineTurnKey.current = "";
   761→    refreshBoardState();
   762→    playSound("move");
   763→  }, [mode, humanColor, status, refreshBoardState]);
   764→
   765→  const resign = useCallback(() => {
   766→    if (status !== "active") return;
   767→    const loser: Side = mode === "ai" ? humanColor : sideOf(turn);
   768→    endGame({ winner: loser === "white" ? "black" : "white", reason: "resignation" });
   769→    toast(`${loser === "white" ? "White" : "Black"} resigned.`, {
   770→      description: "The crown changes hands.",
   771→    });
   772→  }, [status, mode, humanColor, turn, endGame]);
   773→
   774→  const offerDraw = useCallback(() => {
   775→    if (status !== "active") return;
   776→    if (mode === "human") {
   777→      setDrawOfferPending(turn);
   778→      return;
   779→    }
   780→    // AI decides: accepts only if clearly worse and past the opening
   781→    const evalCp = engineInfo?.evalCp ?? 0;
   782→    const aiIsWhite = engineColor === "w";
   783→    const aiEval = aiIsWhite ? evalCp : -evalCp;
   784→    if (aiEval < -180 && history.length >= 24) {
   785→      endGame({ winner: null, reason: "agreement" });
   786→      toast("Draw agreed", { description: "The Grandmaster accepts your offer." });
   787→    } else {
   788→      toast("Draw declined", {
   789→        description: "The Grandmaster smells blood and plays on.",
   790→      });
   791→    }
   792→  }, [status, mode, turn, engineInfo, engineColor, history.length, endGame]);
   793→
   794→  const resolveDrawOffer = useCallback(
   795→    (accept: boolean) => {
   796→      setDrawOfferPending(null);
   797→      if (accept) endGame({ winner: null, reason: "agreement" });
   798→    },
   799→    [endGame]
   800→  );
   801→
   802→  const requestHint = useCallback(async () => {
   803→    if (status !== "active" || thinking || !isHumanTurn) return;
   804→    if (hintsUsed >= 3) {
   805→      toast("No hints left", {
   806→        description: "Three whispers per game — the rest is destiny.",
   807→      });
   808→      return;
   809→    }
   810→    setHintsUsed((h) => h + 1);
   811→    setThinking(true);
   812→    try {
   813→      const res = await fetch("/api/engine", {
   814→        method: "POST",
   815→        headers: { "Content-Type": "application/json" },
   816→        body: JSON.stringify({ fen: gameRef.current.fen(), level: "hard" }),
   817→      });
   818→      if (!res.ok) throw new Error();
   819→      const data = (await res.json()) as { from: Square; to: Square };
   820→      setHint({ from: data.from, to: data.to });
   821→      setSelected(null);
   822→      setLegalForSelected([]);
   823→      window.setTimeout(() => setHint((h) => (h?.from === data.from && h?.to === data.to ? null : h)), 2600);
   824→    } catch {
   825→      toast.error("Hint unavailable right now.");
   826→    } finally {
   827→      setThinking(false);
   828→    }
   829→  }, [status, thinking, isHumanTurn, hintsUsed]);
   830→
   831→  const flipBoard = useCallback(() => {
   832→    setOrientation((o) => (o === "white" ? "black" : "white"));
   833→  }, []);
   834→
   835→  const goToHistory = useCallback((index: number | null) => {
   836→    setViewIndex(index);
   837→    setSelected(null);
   838→    setLegalForSelected([]);
   839→  }, []);
   840→
   841→  const primeAudioOnGesture = useCallback(() => {
   842→    primeAudio();
   843→  }, []);
   844→
   845→  return {
   846→    // state
   847→    fen,
   848→    pieces: displayPieces,
   849→    livePieces: pieces,
   850→    history,
   851→    turn,
   852→    lastMove,
   853→    inCheck,
   854→    checkedSquare,
   855→    selected,
   856→    legalForSelected,
   857→    pendingPromotion,
   858→    status,
   859→    result,
   860→    orientation,
   861→    humanColor,
   862→    engineColor: sideOf(engineColor),
   863→    mode,
   864→    difficulty,
   865→    soundOn,
   866→    showLegal,
   867→    showCoords,
   868→    autoQueen,
   869→    preferredSide,
   870→    clocks,
   871→    activeTimeControl,
   872→    thinking,
   873→    engineInfo,
   874→    hint,
   875→    hintsUsed,
   876→    viewIndex,
   877→    viewing,
   878→    interactive,
   879→    started,
   880→    stats,
   881→    drawOfferPending,
   882→    capturedBy,
   883→    // actions
   884→    selectSquare,
   885→    attemptMove,
   886→    promote,
   887→    cancelPromotion,
   888→    newGame,
   889→    undo,
   890→    resign,
   891→    offerDraw,
   892→    resolveDrawOffer,
   893→    requestHint,
   894→    flipBoard,
   895→    goToHistory,
   896→    setSoundOn,
   897→    setShowLegal,
   898→    setShowCoords,
   899→    setAutoQueen,
   900→    setMode,
   901→    setDifficulty,
   902→    setPreferredSide,
   903→    primeAudioOnGesture,
   904→  };
   905→}
   906→
   907→export type ChessGame = ReturnType<typeof useChessGame>;
